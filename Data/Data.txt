Class WorkoutPlannerAPI:
    “””Class to interact with the Workout Planner API”””
    
    Def __init__(self, api_key):
        Self.api_key = api_key
        Self.host = API_HOST
        Self.base_url = f”https://{self.host}”
        
    Def get_workout_plan(self, time, equipment, muscle, fitness_level, fitness_goals):
        “””
        Fetch a personalized workout plan
        
        Args:
            Time (int): Session duration in minutes
            Equipment (str): Available equipment
            Muscle (str): Target muscle group(s)
            Fitness_level (str): Fitness level (beginner/intermediate/advanced)
            Fitness_goals (str): Fitness goals
            
        Returns:
            Dict: Workout plan data
        “””
        Headers = {
            “X-rapidapi-key”: self.api_key,
            “X-rapidapi-host”: self.host
        }
        
        Params = {
            “Time”: time,
            “Equipment”: equipment,
            “Muscle”: muscle,
            “Fitness_level”: fitness_level,
            “Fitness_goals”: fitness_goals
        }
        
        Try:
            Response = requests.get(
                f”{self.base_url}/customized”,
                params=params,
                headers=headers
            )
            
            If response.status_code == 200:
                Return response.json()
            Else:
                print(f”Error: {response.status_code}”)
                Return None
                
        Except Exception as e:
            print(f”Request failed: {str(e)}”)
            Return None
